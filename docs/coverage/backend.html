// Command server runs the calculator API. It listens on :8080 (or :$PORT)
// and, when STATIC_DIR is set, also serves the built frontend from that
// directory. It shuts down gracefully on SIGINT or SIGTERM.
package main

import (
        "context"
        "fmt"
        "log/slog"
        "net"
        "net/http"
        "os"
        "os/signal"
        "syscall"
        "time"

        "github.com/GabrielHKGodinho/fullstack-calculator/backend/internal/httpapi"
)

const (
        defaultPort     = "8080"
        shutdownTimeout = 10 * time.Second
)

func main() {
        logger := slog.New(slog.NewTextHandler(os.Stderr, nil))
        if err := run(logger); err != nil {
                logger.Error("server failed", "error", err)
                os.Exit(1)
        }
}

// run serves until SIGINT or SIGTERM, then shuts down gracefully.
func run(logger *slog.Logger) error {
        port := os.Getenv("PORT")
        if port == "" {
                port = defaultPort
        }
        staticDir := os.Getenv("STATIC_DIR")

        ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
        defer stop()

        srv := newServer(":"+port, newHandler(logger, staticDir))

        // Listen before logging, so "listening" is only logged once the port is
        // actually bound (a busy port fails here instead).
        ln, err := net.Listen("tcp", srv.Addr)
        if err != nil {
                return err
        }
        logger.Info("listening", "addr", ln.Addr().String(), "static_dir", staticDir)

        serveErr := make(chan error, 1)
        go func() { serveErr <- srv.Serve(ln) }()

        select {
        case err := <-serveErr:
                return err
        case <-ctx.Done():
        }

        // Restore default signal handling, so a second Ctrl-C kills the process
        // right away instead of waiting for the shutdown.
        stop()
        logger.Info("shutting down")

        shutdownCtx, cancel := context.WithTimeout(context.Background(), shutdownTimeout)
        defer cancel()
        if err := srv.Shutdown(shutdownCtx); err != nil {
                return fmt.Errorf("shutdown: %w", err)
        }
        logger.Info("server stopped")
        return nil
}

// newHandler routes /api/ to the API. When staticDir is set, every other path
// is served from that directory; otherwise those paths get net/http's plain
// 404.
func newHandler(logger *slog.Logger, staticDir string) http.Handler {
        mux := http.NewServeMux()
        mux.Handle("/api/", httpapi.New(logger))
        if staticDir != "" {
                mux.Handle("/", http.FileServer(http.Dir(staticDir)))
        }
        return mux
}

// newServer returns an http.Server with explicit timeouts, so slow or idle
// clients cannot hold connections open forever.
func newServer(addr string, h http.Handler) *http.Server {
        return &http.Server{
                Addr:              addr,
                Handler:           h,
                ReadHeaderTimeout: 5 * time.Second,
                ReadTimeout:       10 * time.Second,
                WriteTimeout:      10 * time.Second,
                IdleTimeout:       60 * time.Second,
        }
}

// Package calculator implements the calculator's operations. It knows nothing
// about HTTP: callers look up an operation by name, apply it to its operands
// and check the returned error against the sentinel errors below.
package calculator

import (
        "errors"
        "math"
)

// Sentinel errors returned by Lookup and Operation.Apply. Check them with
// errors.Is.
var (
        ErrUnknownOperation  = errors.New("unknown operation")
        ErrWrongOperandCount = errors.New("wrong number of operands")
        ErrDivisionByZero    = errors.New("division by zero")
        ErrSqrtOfNegative    = errors.New("square root of a negative number")
        ErrResultNotFinite   = errors.New("result is not finite")
)

// Operation is a named calculator operation that takes Arity operands.
// Values are obtained from Lookup; the zero value is not a valid operation.
type Operation struct {
        Name  string
        Arity int
        fn    func(x []float64) (float64, error)
}

var operations = map[string]Operation{
        "add":        {Name: "add", Arity: 2, fn: add},
        "subtract":   {Name: "subtract", Arity: 2, fn: subtract},
        "multiply":   {Name: "multiply", Arity: 2, fn: multiply},
        "divide":     {Name: "divide", Arity: 2, fn: divide},
        "power":      {Name: "power", Arity: 2, fn: power},
        "sqrt":       {Name: "sqrt", Arity: 1, fn: sqrt},
        "percentage": {Name: "percentage", Arity: 2, fn: percentage},
}

// Lookup returns the operation with the given name. Names are matched
// exactly (case-sensitive). An unknown name returns ErrUnknownOperation.
func Lookup(name string) (Operation, error) {
        op, ok := operations[name]
        if !ok {
                return Operation{}, ErrUnknownOperation
        }
        return op, nil
}

// Apply runs the operation on operands. It returns ErrWrongOperandCount if
// len(operands) is not the operation's arity, an operation-specific error
// (ErrDivisionByZero, ErrSqrtOfNegative), or ErrResultNotFinite if the
// result is ±Inf or NaN. A non-nil error always comes with a zero result.
func (op Operation) Apply(operands []float64) (float64, error) {
        if op.fn == nil {
                return 0, ErrUnknownOperation
        }
        if len(operands) != op.Arity {
                return 0, ErrWrongOperandCount
        }
        result, err := op.fn(operands)
        if err != nil {
                return 0, err
        }
        // JSON cannot encode ±Inf or NaN, so none may leave this package.
        if math.IsInf(result, 0) || math.IsNaN(result) {
                return 0, ErrResultNotFinite
        }
        return result, nil
}

func add(x []float64) (float64, error)      { return x[0] + x[1], nil }
func subtract(x []float64) (float64, error) { return x[0] - x[1], nil }
func multiply(x []float64) (float64, error) { return x[0] * x[1], nil }
func power(x []float64) (float64, error)    { return math.Pow(x[0], x[1]), nil }

func divide(x []float64) (float64, error) {
        // b == 0 is also true for -0, so 1/-0 and 0/0 are rejected here too.
        if x[1] == 0 {
                return 0, ErrDivisionByZero
        }
        return x[0] / x[1], nil
}

func sqrt(x []float64) (float64, error) {
        // -0 is not < 0, so sqrt(-0) = -0 is allowed, as in math.Sqrt.
        if x[0] < 0 {
                return 0, ErrSqrtOfNegative
        }
        return math.Sqrt(x[0]), nil
}

// percentage returns a% of b.
func percentage(x []float64) (float64, error) { return x[0] * x[1] / 100, nil }

// Package httpapi exposes the calculator as a JSON REST API. It owns
// everything HTTP: routing, the method check, body decoding and validation,
// mapping calculator errors to status codes, and writing JSON responses.
package httpapi

import (
        "errors"
        "fmt"
        "log/slog"
        "net/http"

        "github.com/GabrielHKGodinho/fullstack-calculator/backend/internal/calculator"
)

// New returns the API handler. It serves POST /api/v1/{operation} and answers
// every other path under /api/ with a JSON 404. logger is used only for
// unexpected errors and must not be nil.
func New(logger *slog.Logger) http.Handler {
        h := handler{logger: logger}
        mux := http.NewServeMux()
        // The patterns have no method on purpose: handleOperation checks the
        // method itself so that the 405 response has a JSON body.
        mux.HandleFunc("/api/v1/{operation}", h.handleOperation)
        mux.HandleFunc("/api/", h.handleNotFound)
        return mux
}

type handler struct {
        logger *slog.Logger
}

// handleOperation runs the checks of SPEC §4.4 in order; the first failure
// wins.
func (h handler) handleOperation(w http.ResponseWriter, r *http.Request) {
        name := r.PathValue("operation")
        op, err := calculator.Lookup(name)
        if err != nil {
                writeError(w, &apiError{
                        status:  http.StatusNotFound,
                        code:    codeUnknownOperation,
                        message: fmt.Sprintf("unknown operation %q", name),
                })
                return
        }

        if r.Method != http.MethodPost {
                w.Header().Set("Allow", http.MethodPost)
                writeError(w, &apiError{
                        status:  http.StatusMethodNotAllowed,
                        code:    codeMethodNotAllowed,
                        message: fmt.Sprintf("method %s not allowed, use POST", r.Method),
                })
                return
        }

        req, apiErr := decodeRequest(w, r)
        if apiErr != nil {
                writeError(w, apiErr)
                return
        }

        operands, apiErr := operandValues(req.Operands)
        if apiErr != nil {
                writeError(w, apiErr)
                return
        }

        result, err := op.Apply(operands)
        if err != nil {
                writeError(w, h.calculatorError(op, len(operands), err))
                return
        }

        writeJSON(w, http.StatusOK, resultResponse{Result: result})
}

func (h handler) handleNotFound(w http.ResponseWriter, r *http.Request) {
        writeError(w, &apiError{
                status:  http.StatusNotFound,
                code:    codeNotFound,
                message: "not found",
        })
}

// calculatorError maps an error from op.Apply, called with n operands, to an
// API error. An error it does not know is a bug: it is logged and becomes a
// 500.
func (h handler) calculatorError(op calculator.Operation, n int, err error) *apiError {
        switch {
        case errors.Is(err, calculator.ErrWrongOperandCount):
                noun := "operands"
                if op.Arity == 1 {
                        noun = "operand"
                }
                return &apiError{
                        status:  http.StatusBadRequest,
                        code:    codeWrongOperandCount,
                        message: fmt.Sprintf("%s takes %d %s, got %d", op.Name, op.Arity, noun, n),
                }
        case errors.Is(err, calculator.ErrDivisionByZero):
                return &apiError{
                        status:  http.StatusUnprocessableEntity,
                        code:    codeDivisionByZero,
                        message: "cannot divide by zero",
                }
        case errors.Is(err, calculator.ErrSqrtOfNegative):
                return &apiError{
                        status:  http.StatusUnprocessableEntity,
                        code:    codeSqrtOfNegative,
                        message: "cannot take the square root of a negative number",
                }
        case errors.Is(err, calculator.ErrResultNotFinite):
                return &apiError{
                        status:  http.StatusUnprocessableEntity,
                        code:    codeResultNotFinite,
                        message: "result is too large or undefined",
                }
        default:
                h.logger.Error("unexpected calculator error", "operation", op.Name, "error", err)
                return &apiError{
                        status:  http.StatusInternalServerError,
                        code:    codeInternalError,
                        message: "internal server error",
                }
        }
}

package httpapi

import (
        "encoding/json"
        "errors"
        "fmt"
        "io"
        "net/http"
        "strings"
)

// maxBodyBytes is the largest request body accepted (1 MB).
const maxBodyBytes = 1 << 20

// calculateRequest is the body of POST /api/v1/{operation}.
type calculateRequest struct {
        // Pointers keep a JSON null element distinguishable from 0.
        Operands []*float64 `json:"operands"`
}

// decodeRequest decodes the request body, which must be a single JSON object
// of at most maxBodyBytes with no field other than "operands". Only
// whitespace may follow the object.
//
// Errors: BODY_TOO_LARGE (413), UNKNOWN_FIELD (400), and INVALID_JSON (400)
// for everything else (empty body, invalid syntax, wrong types, numbers
// outside the float64 range, trailing data).
func decodeRequest(w http.ResponseWriter, r *http.Request) (calculateRequest, *apiError) {
        dec := json.NewDecoder(http.MaxBytesReader(w, r.Body, maxBodyBytes))
        dec.DisallowUnknownFields()

        var req calculateRequest
        if err := dec.Decode(&req); err != nil {
                if e := bodyTooLarge(err); e != nil {
                        return calculateRequest{}, e
                }
                if e := unknownField(err); e != nil {
                        return calculateRequest{}, e
                }
                return calculateRequest{}, invalidJSON()
        }

        // Anything but whitespace after the object is trailing data. A
        // RawMessage accepts any value, so a second object counts as trailing
        // data rather than as an unknown field.
        var extra json.RawMessage
        if err := dec.Decode(&extra); !errors.Is(err, io.EOF) {
                if e := bodyTooLarge(err); e != nil {
                        return calculateRequest{}, e
                }
                return calculateRequest{}, invalidJSON()
        }

        return req, nil
}

// bodyTooLarge returns a BODY_TOO_LARGE error if err comes from the
// http.MaxBytesReader limit, and nil otherwise.
func bodyTooLarge(err error) *apiError {
        var maxBytesErr *http.MaxBytesError
        if !errors.As(err, &maxBytesErr) {
                return nil
        }
        return &apiError{
                status:  http.StatusRequestEntityTooLarge,
                code:    codeBodyTooLarge,
                message: "request body must not exceed 1MB",
        }
}

// unknownField returns an UNKNOWN_FIELD error if err comes from
// DisallowUnknownFields, and nil otherwise. encoding/json has no typed error
// for this case, so its message is the only way to tell it apart.
func unknownField(err error) *apiError {
        field, ok := strings.CutPrefix(err.Error(), "json: unknown field ")
        if !ok {
                return nil
        }
        return &apiError{
                status:  http.StatusBadRequest,
                code:    codeUnknownField,
                message: fmt.Sprintf("unknown field %s", field), // field is already quoted
        }
}

func invalidJSON() *apiError {
        return &apiError{
                status:  http.StatusBadRequest,
                code:    codeInvalidJSON,
                message: "request body must be valid JSON",
        }
}

// operandValues checks that operands is present and has no null elements,
// and returns its values. The operand count is checked by calculator.
func operandValues(raw []*float64) ([]float64, *apiError) {
        if raw == nil {
                return nil, &apiError{
                        status:  http.StatusBadRequest,
                        code:    codeMissingOperands,
                        message: `"operands" is required`,
                }
        }
        values := make([]float64, len(raw))
        for i, p := range raw {
                if p == nil {
                        return nil, &apiError{
                                status:  http.StatusBadRequest,
                                code:    codeNullOperand,
                                message: "operands must not be null",
                        }
                }
                values[i] = *p
        }
        return values, nil
}

package httpapi

import (
        "encoding/json"
        "net/http"
)

// Error codes are the stable API contract; messages are for humans and may
// change.
const (
        codeInvalidJSON       = "INVALID_JSON"
        codeUnknownField      = "UNKNOWN_FIELD"
        codeMissingOperands   = "MISSING_OPERANDS"
        codeNullOperand       = "NULL_OPERAND"
        codeWrongOperandCount = "WRONG_OPERAND_COUNT"
        codeUnknownOperation  = "UNKNOWN_OPERATION"
        codeNotFound          = "NOT_FOUND"
        codeMethodNotAllowed  = "METHOD_NOT_ALLOWED"
        codeBodyTooLarge      = "BODY_TOO_LARGE"
        codeDivisionByZero    = "DIVISION_BY_ZERO"
        codeSqrtOfNegative    = "SQRT_OF_NEGATIVE"
        codeResultNotFinite   = "RESULT_NOT_FINITE"
        codeInternalError     = "INTERNAL_ERROR"
)

// resultResponse is the body of a successful calculation.
type resultResponse struct {
        Result float64 `json:"result"`
}

// errorResponse is the body of every error response.
type errorResponse struct {
        Error errorBody `json:"error"`
}

type errorBody struct {
        Code    string `json:"code"`
        Message string `json:"message"`
}

// apiError is a failed request: the HTTP status plus the code and message
// that go into the error body.
type apiError struct {
        status  int
        code    string
        message string
}

// writeJSON writes v as a JSON response with the given status.
func writeJSON(w http.ResponseWriter, status int, v any) {
        w.Header().Set("Content-Type", "application/json")
        w.WriteHeader(status)
        // The values written here are strings and finite floats, so encoding
        // can only fail if the client has gone away. There is nothing useful
        // to do about that, so the error is ignored.
        _ = json.NewEncoder(w).Encode(v)
}

// writeError writes e as a JSON error response.
func writeError(w http.ResponseWriter, e *apiError) {
        writeJSON(w, e.status, errorResponse{Error: errorBody{Code: e.code, Message: e.message}})
}
